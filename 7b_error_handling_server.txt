from flask import Flask, request, render_template

from EmotionDetection.emotion_detection import emotion_detector

app = Flask(
    "Emotion Detector", template_folder="../templates", static_folder="../static"
)


@app.route("/emotionDetector")
def find_emotion_scores():
    text_to_analyze = request.args.get("textToAnalyze")
    r = emotion_detector(text_to_analyze)
    if r["dominant_emotion"] == None:
        return "Invalid text! Please try again!."

    return f"For the given statement, the system response is 'anger': {r['anger']}, 'disgust': {r['disgust']}, 'fear': {r['fear']}, 'joy': {r['joy']} and 'sadness': {r['sadness']}. The dominant emotion is {r['dominant_emotion']}."


@app.route("/")
def render_index_page():
    return render_template("index.html")

if __name__ == "__main__":
    app.run(port=5000)
